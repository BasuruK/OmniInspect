package mcpserver

import (
	"context"
	"io"
	"net"
	"net/http"
	"strings"
	"testing"
	"time"
)

// Archived diagnostic: copy into internal/adapter/mcpserver to run, then remove.
// This intentionally fails against a server that loses sessions on restart.
func TestDiagnosticRestartRetainsClientSession(t *testing.T) {
	deps, cleanup := testDeps(t)
	defer cleanup()
	const token = "diagnostic-not-a-real-secret"
	request := func(addr, session, body string) (int, string) {
		t.Helper()
		req, err := http.NewRequest(http.MethodPost, "http://"+addr, strings.NewReader(body))
		if err != nil { t.Fatal(err) }
		req.Header.Set("Authorization", "Bearer "+token)
		req.Header.Set("Content-Type", "application/json")
		req.Header.Set("Accept", "application/json, text/event-stream")
		if session != "" { req.Header.Set("Mcp-Session-Id", session) }
		resp, err := (&http.Client{Timeout: 3*time.Second}).Do(req)
		if err != nil { t.Fatal(err) }
		defer resp.Body.Close()
		_, _ = io.Copy(io.Discard, resp.Body)
		return resp.StatusCode, resp.Header.Get("Mcp-Session-Id")
	}
	boot := func(addr string) (string, func()) {
		t.Helper()
		ln, err := net.Listen("tcp", addr)
		if err != nil { t.Fatal(err) }
		ctx, cancel := context.WithCancel(context.Background())
		done := make(chan error, 1)
		go func() { done <- NewServer(deps).ServeStreamableHTTP(ctx, ln, token) }()
		return ln.Addr().String(), func() {
			cancel()
			select {
			case err := <-done: if err != nil { t.Error(err) }
			case <-time.After(6*time.Second): t.Fatal("shutdown timeout")
			}
		}
	}
	const initialize = `{"jsonrpc":"2.0","id":1,"method":"initialize","params":{"protocolVersion":"2025-03-26","capabilities":{},"clientInfo":{"name":"restart-probe","version":"test"}}}`
	addr, stop := boot("127.0.0.1:0")
	status, session := request(addr, "", initialize)
	if status != http.StatusOK { stop(); t.Fatalf("initialize status=%d", status) }
	if session == "" { session = "pre-restart-session" }
	stop()
	addr, stop = boot(addr)
	defer stop()
	status, _ = request(addr, session, `{"jsonrpc":"2.0","id":2,"method":"ping"}`)
	freshStatus, _ := request(addr, "", initialize)
	t.Logf("same bearer after restart: old session HTTP %d; fresh initialize HTTP %d", status, freshStatus)
	if freshStatus != http.StatusOK { t.Fatalf("unchanged bearer rejected: HTTP %d", freshStatus) }
	if status != http.StatusOK { t.Fatalf("IDE session requires reinitialization after restart: HTTP %d", status) }
}
